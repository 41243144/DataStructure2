## 題目描述

請修改第八章講義 Chaining implementation 程式，增加一函式 `int search(int key)`，搜尋 key 值是否在 Hash table 中，若存在則傳回 Hash table index，否則傳回 -1。

---

## 程式碼

```cpp
#include <iostream>
using namespace std;

#define b 13                // 定義 hash table bucket 數
#define EMPTY -32768        // 定義空節點標記

typedef struct node {
    int key;
    struct node* next;
} list_node;

// Hash table 由 b 個鏈結串列所組成
list_node* ht[b];

// 初始化，每個 bucket 分配一個 head 節點
void initialize() {
    for (int i = 0; i < b; i++) {
        ht[i] = new list_node;
        ht[i]->key = EMPTY;
        ht[i]->next = NULL;
    }
}

// 簡單取餘數作為 hash function
int hash_fun(int key) {
    return key % b;
}

// 插入新鍵值到對應的鏈結串列尾端
void insert(int key) {
    int address = hash_fun(key);
    list_node* ptr = new list_node;
    ptr->key = key;
    ptr->next = NULL;

    list_node* tmp = ht[address];
    while (tmp->next != NULL)
        tmp = tmp->next;
    tmp->next = ptr;
}

// 搜尋函式：若找到 key，回傳 bucket index；否則回傳 -1
int search(int key) {
    int address = hash_fun(key);
    list_node* tmp = ht[address]->next;  // 跳過 head 空節點
    while (tmp != nullptr) {
        if (tmp->key == key)
            return address;
        tmp = tmp->next;
    }
    return -1;
}

// 顯示整個 hash table 的內容
void show() {
    for (int i = 0; i < b; i++) {
        cout << "ht[" << i << "]\t";
        list_node* tmp = ht[i];
        do {
            if (tmp->key != EMPTY)
                cout << tmp->key << "\t";
            tmp = tmp->next;
        } while (tmp != NULL);
        cout << endl;
    }
}

int main() {
    initialize();
    // 範例插入
    insert(13); insert(14); insert(26);
    insert(60); insert(39); insert(40);
    insert(86); insert(15); insert(25);

    show();

    int key;
    while (cin >> key) {
        cout << "---------------------------------------" << endl;
        cout << "位置在: " << search(key) << endl;
    }
    return 0;
}
```
